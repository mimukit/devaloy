```python
import asyncio, os, sys, asyncssh
D = sys.argv[1]
class Relay(asyncssh.SSHTCPSession):
    def __init__(self, port): self.port = port; self.w = None; self.buf = []
    def connection_made(self, chan):
        self.chan = chan
        asyncio.ensure_future(self.open())
    async def open(self):
        try:
            r, self.w = await asyncio.open_connection("127.0.0.1", self.port)
        except OSError:
            self.chan.close(); return
        for b in self.buf: self.w.write(b)
        while True:
            d = await r.read(65536)
            if not d: break
            self.chan.write(d)
        self.chan.write_eof(); self.chan.close()
    def data_received(self, data, datatype):
        if self.w: self.w.write(data)
        else: self.buf.append(data)
    def eof_received(self):
        if self.w: self.w.write_eof()
class S(asyncssh.SSHServer):
    def begin_auth(self, u): return False
    def connection_requested(self, h, p, oh, op):
        return Relay(p + 10000)  # the "box" server for port p listens on p+10000
async def handle(process):
    env = dict(os.environ, PATH=D + "/boxbin:" + os.environ["PATH"])
    p = await asyncio.create_subprocess_shell(process.command, stdout=-1, stderr=-1, env=env)
    out, err = await p.communicate()
    process.stdout.write(out.decode()); process.stderr.write(err.decode()); process.exit(p.returncode)
async def main():
    key = asyncssh.generate_private_key("ssh-ed25519")
    await asyncssh.create_server(S, "127.0.0.1", 2222, server_host_keys=[key], process_factory=handle)
    print("ready", flush=True)
    await asyncio.Future()
asyncio.run(main())
```
